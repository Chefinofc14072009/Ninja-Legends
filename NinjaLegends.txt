​-- Função Rigorosa: Só equipa se for uma Espada real de ownedSwords (Ignora Shurikens)
local function equipOwnedSwordOnly(character)
if not character then return end
local humanoid = character:FindFirstChildOfClass("Humanoid")
local equippedTool = character:FindFirstChildOfClass("Tool")
​pcall(function()
local ownedSwords = LocalPlayer:FindFirstChild("ownedSwords")
if not ownedSwords then return end
​if equippedTool and ownedSwords:FindFirstChild(equippedTool.Name) then
return
end
​local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
if not backpack then return end
​for _, swordFolder in ipairs(ownedSwords:GetChildren()) do
local swordName = swordFolder.Name
local toolInBackpack = backpack:FindFirstChild(swordName)
​if toolInBackpack and humanoid then
humanoid:EquipTool(toolInBackpack)
return
end
end
end)
end
